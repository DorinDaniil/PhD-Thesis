\documentclass[tikz,border=2pt]{standalone}
\usepackage[T2A]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[russian]{babel}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta,positioning,calc}

% цвета строк
\definecolor{fillA}{RGB}{232,238,247} \definecolor{lineA}{RGB}{70,110,160}  \definecolor{darkA}{RGB}{200,215,235}
\definecolor{fillB}{RGB}{240,238,222} \definecolor{lineB}{RGB}{120,120,70}  \definecolor{darkB}{RGB}{214,210,170}
\definecolor{fillC}{RGB}{248,230,226} \definecolor{lineC}{RGB}{170,70,60}   \definecolor{darkC}{RGB}{236,206,198}
\definecolor{lagc}{RGB}{70,110,160}
\definecolor{soft}{RGB}{110,110,110}

% Кегль надписей. Собственная ширина рисунка близка к ширине полосы диссертации (17,5 см), поэтому при
% вставке на всю полосу надписи почти не уменьшаются: 12 pt в исходнике дают около 11,5 pt на странице.
% Чтобы уместить такой кегль, столбцы имеют разную ширину: узкий для входа, широкий для метода,
% узкий для результата; надписи результата набраны в две строки.
\newcommand{\figfont}{\fontsize{12}{14.5}\selectfont}

\begin{document}
\begin{tikzpicture}[
  font=\figfont, line cap=round, line join=round,
  >={Stealth[length=6pt,width=4.5pt]},
  box/.style={draw, rounded corners=3pt, line width=0.8pt, minimum height=1.76cm, align=center, inner sep=3pt},
  colL/.style={box, minimum width=3.7cm, text width=3.4cm},
  colM/.style={box, minimum width=5.65cm, text width=5.35cm},
  colR/.style={box, minimum width=3.25cm, text width=2.95cm},
  rowlbl/.style={align=right, text width=3.3cm, inner sep=0pt},
  arr/.style={->, line width=0.8pt},
  lag/.style={->, line width=0.8pt, lagc, densely dashed},
  lagline/.style={line width=0.8pt, lagc, densely dashed},
  laglbl/.style={text=lagc, fill=white, inner sep=1.5pt},
]

% ---------- строка I: временная динамика ----------
\node[colL, fill=fillA, draw=lineA] (a1) at (0,0) {стимул};
\node[colM, fill=fillA, draw=lineA, right=0.4cm of a1] (a2) {линейная\\авторегрессионная\\модель прогноза сигнала};
\node[colR, fill=darkA, draw=lineA, right=0.4cm of a2] (a3) {оценка\\лага $\Delta t$};
\draw[arr, lineA] (a1) -- (a2); \draw[arr, lineA] (a2) -- (a3);
\node[rowlbl, left=0.3cm of a1] {I. Временная\\динамика};

% ---------- строка II: пространство и время ----------
\node[colL, fill=fillB, draw=lineB, below=0.95cm of a1] (b1) {многоканальный\\временной ряд};
\node[colM, fill=fillB, draw=lineB, right=0.4cm of b1] (b2) {маски информативных\\каналов, риманова\\геометрия ковариаций};
\node[colR, fill=darkB, draw=lineB, right=0.4cm of b2] (b3) {категория\\стимула};
\draw[arr, lineB] (b1) -- (b2); \draw[arr, lineB] (b2) -- (b3);
\node[rowlbl, left=0.3cm of b1] {II. Пространство\\и время};

% ---------- строка III: мультимодальность ----------
\node[colL, fill=fillC, draw=lineC, below=0.95cm of b1] (c1) {две синхронно\\регистрируемые\\модальности};
\node[colM, fill=fillC, draw=lineC, right=0.4cm of c1] (c2) {восстановление каналов,\\кодировщики, объединение\\представлений};
\node[colR, fill=darkC, draw=lineC, right=0.4cm of c2] (c3) {реконструкция\\стимула};
\draw[arr, lineC] (c1) -- (c2); \draw[arr, lineC] (c2) -- (c3);
\node[rowlbl, left=0.3cm of c1] {III. Мульти-\\модальность};

% ---------- передача лага ----------
\coordinate (r) at ($(a3.east)+(0.4,0)$);
\coordinate (b2u) at ($(b2.south)+(0,-0.5)$);
\coordinate (c1u) at ($(c1.south)+(0,-0.5)$);
\draw[lagline] (a3.east) -- (r) -- (r |- b2u) -- (b2u);
\draw[lag] (b2u) -- (b2.south);
\node[laglbl, anchor=west] at ($(b2u)!0.5!(b2.south)+(0.12,0)$) {$\Delta t$};
\draw[lagline] (r |- b2u) -- (r |- c1u) -- (c1u);
\draw[lag] (c1u) -- (c1.south);
\node[laglbl, anchor=west] at ($(c1u)!0.5!(c1.south)+(0.12,0)$) {$\Delta t$};

% ---------- ось усложнения ----------
\coordinate (top) at ($(a1.north west)+(-3.85,0)$);
\coordinate (bot) at ($(c1.south west)+(-3.85,0)$);
\draw[->, soft, line width=0.8pt] (top) -- (bot);
\node[rotate=90, text=soft, anchor=south, inner sep=2pt] at ($(top)!0.5!(bot)$) {усложнение учитываемой структуры};

\end{tikzpicture}
\end{document}
